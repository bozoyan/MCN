## 概述

本教程说明如何将新的独立界面（如 `news.py`）整合到 `storyboard_generator.py` 主程序中，使其成为左侧导航菜单的一个子页面。

### 核心原则

- ✅ **使用 QWidget 类**：新界面应该是继承自 `QWidget` 的页面类，**不是**独立的 `QMainWindow`
- ✅ **接收 parent 参数**：构造函数必须接收 `parent` 参数，用于访问主窗口
- ❌ **不要创建独立 QApplication**：不要在自己的文件中创建 `QApplication` 和事件循环
- ✅ **使用项目通用功能**：复用项目中的常用功能模块（消息提示、图片处理、API 调用等）

本版本新增以下常用功能模块的详细说明：

### 自动消失的消息提示

使用 `InfoBar` 替代传统的 `QMessageBox`，消息会在指定时间后自动消失，无需用户点击确定。

#### 实现代码

```python
from qfluentwidgets import InfoBar, InfoBarPosition, InfoBarIcon

def show_auto_hide_message(parent, title, content, severity='info', duration=3000):
    """显示自动消失的 InfoBar 消息

    Args:
        parent: 父窗口或 Widget
        title: 消息标题
        content: 消息内容
        severity: 消息类型 ('info', 'success', 'warning', 'error')
        duration: 显示时长（毫秒），默认 3 秒
    """
    # 根据 severity 选择图标和类型
    icon_config = {
        'info': (InfoBarIcon.INFORMATION, InfoBarPosition.TOP),
        'success': (InfoBarIcon.SUCCESS, InfoBarPosition.TOP),
        'warning': (InfoBarIcon.WARNING, InfoBarPosition.TOP),
        'error': (InfoBarIcon.ERROR, InfoBarPosition.TOP)
    }

    icon, position = icon_config.get(severity, icon_config['info'])

    # 创建 InfoBar
    info_bar = InfoBar(
        icon,
        title,
        content,
        parent=parent,
        position=position,
        duration=duration
    )
    info_bar.show()
```

#### 使用示例

```python
# 成功消息
show_auto_hide_message(self, "成功", "文件已保存", "success")

# 警告消息
show_auto_hide_message(self, "警告", "文件大小超过限制", "warning")

# 错误消息
show_auto_hide_message(self, "错误", "网络连接失败", "error")

# 信息消息
show_auto_hide_message(self, "提示", "正在处理...", "info", 5000)  # 5秒后消失
```

#### 对比传统 QMessageBox

| 传统方式 | 新方式 |
|---------|--------|
| `QMessageBox.information(self, "成功", "完成")` | `show_auto_hide_message(self, "成功", "完成", "success")` |
| `QMessageBox.warning(self, "警告", "注意")` | `show_auto_hide_message(self, "警告", "注意", "warning")` |
| `QMessageBox.critical(self, "错误", "失败")` | `show_auto_hide_message(self, "错误", "失败", "error")` |
| 需要点击"确定" | 自动消失 |
| 阻塞程序执行 | 不阻塞 |

### 支持拖拽的图片标签

支持拖拽本地文件和网络 URL 的图片预览标签。

#### 实现代码

```python
from PyQt5.QtWidgets import QLabel
from PyQt5.QtCore import Qt, pyqtSignal
from PyQt5.QtGui import QPixmap

class DragDropImageLabel(QLabel):
    """支持拖拽本地图片和网络URL的图片预览标签"""

    image_dropped = pyqtSignal(str)  # 发射信号：文件路径或URL

    def __init__(self, parent=None):
        super().__init__(parent)
        self.setAcceptDrops(True)
        self.setAlignment(Qt.AlignCenter)
        self.setFixedSize(400, 600)
        self.setStyleSheet("border: 2px dashed #ccc; border-radius: 8px; background: #1E1E1E;")
        self.setText("暂无图片\n\n支持拖拽本地图片或网络 URL")

    def dragEnterEvent(self, event):
        """拖拽进入事件"""
        if event.mimeData().hasUrls() or event.mimeData().hasText():
            event.acceptProposedAction()
            self.setStyleSheet("border: 2px dashed #6699ff; border-radius: 8px; background: #e6f0ff;")
        else:
            event.ignore()

    def dragLeaveEvent(self, event):
        """拖拽离开事件"""
        event.accept()
        self.setStyleSheet("border: 2px dashed #ccc; border-radius: 8px; background: #1E1E1E;")

    def dropEvent(self, event):
        """拖拽放下事件"""
        self.setStyleSheet("border: 2px dashed #ccc; border-radius: 8px; background: #1E1E1E;")

        # 检查是否有URL（支持拖拽文件）
        if event.mimeData().hasUrls():
            urls = event.mimeData().urls()
            if urls:
                url = urls[0]
                if url.isLocalFile():
                    # 本地文件
                    local_path = url.toLocalFile()
                    image_extensions = ['.png', '.jpg', '.jpeg', '.gif', '.bmp', '.webp']
                    if any(local_path.lower().endswith(ext) for ext in image_extensions):
                        self.image_dropped.emit(local_path)
                    else:
                        show_auto_hide_message(self.window(), "警告", "请拖拽图片文件", "warning")
                else:
                    # 网络URL
                    network_url = url.toString()
                    self.image_dropped.emit(network_url)
                event.acceptProposedAction()
        # 检查是否有文本（支持拖拽URL文本）
        elif event.mimeData().hasText():
            text = event.mimeData().text().strip()
            if text.startswith(('http://', 'https://')):
                self.image_dropped.emit(text)
                event.acceptProposedAction()
```

#### 使用示例

```python
class MyPage(QWidget):
    def __init__(self, parent=None):
        super().__init__(parent)

        # 创建拖拽图片标签
        self.image_label = DragDropImageLabel()

        # 连接信号
        self.image_label.image_dropped.connect(self.on_image_dropped)

        # 添加到布局
        layout = QVBoxLayout(self)
        layout.addWidget(self.image_label)

    def on_image_dropped(self, path_or_url):
        """处理图片拖拽事件"""
        if path_or_url.startswith(('http://', 'https://')):
            # 网络URL - 下载并处理
            self.download_and_process_image(path_or_url)
        else:
            # 本地文件 - 直接处理
            self.process_local_image(path_or_url)
```

### 图片处理与 WebP 转换

统一的图片处理流程：本地/远程图片 → 转换为 WebP → 保存到 `output/up/` → 转换为 Base64

#### 实现代码

```python
import os
from datetime import datetime
from io import BytesIO
from PIL import Image
import base64
import requests

def save_image_as_webp(img, source_path, is_network=False):
    """将图片转换为 webp 并保存到 output/up/ 文件夹

    Args:
        img: PIL Image 对象
        source_path: 源路径（本地路径或网络URL）
        is_network: 是否为网络图片

    Returns:
        tuple: (webp_abs_path, data_url)
    """
    # 确保 output/up/ 目录存在
    output_dir = os.path.join("output", "up")
    os.makedirs(output_dir, exist_ok=True)

    # 生成时间戳文件名
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S_%f')[:-3]  # 精确到毫秒
    webp_filename = f"{timestamp}.webp"
    webp_path = os.path.join(output_dir, webp_filename)
    webp_abs_path = os.path.abspath(webp_path)

    # 转换为 RGB 模式（WebP 不支持 RGBA）
    if img.mode in ('RGBA', 'LA', 'P'):
        # 创建白色背景
        background = Image.new('RGB', img.size, (255, 255, 255))
        if img.mode == 'P':
            img = img.convert('RGBA')
        if img.mode in ('RGBA', 'LA'):
            background.paste(img, mask=img.split()[-1] if img.mode == 'RGBA' else None)
            img = background
        else:
            img = img.convert('RGB')
    elif img.mode != 'RGB':
        img = img.convert('RGB')

    # 保存为 WebP 格式（99% 画质）
    img.save(webp_abs_path, format='WebP', quality=99, method=6)

    # 读取 webp 文件并转换为 base64
    with open(webp_abs_path, 'rb') as f:
        webp_data = f.read()

    base64_data = base64.b64encode(webp_data).decode('utf-8')

    # 验证 Base64 长度（必须是 4 的倍数）
    base64_length = len(base64_data)
    padding_needed = (4 - base64_length % 4) % 4
    if padding_needed > 0:
        base64_data += '=' * padding_needed

    # 创建 data URL
    data_url = f"data:image/webp;base64,{base64_data}"

    return webp_abs_path, data_url


def process_local_image(file_path):
    """处理本地图片"""
    abs_path = os.path.abspath(file_path)
    filename = os.path.basename(abs_path)

    try:
        # 使用 Pillow 打开图片
        img = Image.open(abs_path)

        # 转换为 webp 并保存
        webp_path, data_url = save_image_as_webp(img, abs_path, is_network=False)

        return webp_path, data_url
    except Exception as e:
        show_auto_hide_message(None, "错误", f"无法处理图片: {str(e)}", "error")
        return None, None


def process_network_image(url):
    """处理网络图片"""
    try:
        # 下载图片
        response = requests.get(url, timeout=30)
        response.raise_for_status()

        # 使用 Pillow 打开图片
        img = Image.open(BytesIO(response.content))

        # 转换为 webp 并保存
        webp_path, data_url = save_image_as_webp(img, url, is_network=True)

        return webp_path, data_url
    except Exception as e:
        show_auto_hide_message(None, "错误", f"无法下载图片: {str(e)}", "error")
        return None, None
```

#### 使用示例

### 远程 API 调用与文件下载

标准化的 API 调用流程，支持远程文件下载到本地。

#### 实现代码

```python
import os
import requests
import json
from datetime import datetime

def download_remote_file(remote_url, output_dir="output"):
    """下载远程文件到本地

    Args:
        remote_url: 远程文件 URL
        output_dir: 保存目录，默认为 "output"

    Returns:
        str: 本地文件路径，失败返回 None
    """
    try:
        # 确保输出目录存在
        os.makedirs(output_dir, exist_ok=True)

        # 从远程 URL 中提取原始文件名
        original_filename = os.path.basename(remote_url)
        if not original_filename or original_filename == '':
            # 如果无法提取文件名，使用默认命名
            original_filename = f"download_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"

        # 构建保存路径
        save_path = os.path.join(output_dir, original_filename)

        # 下载文件
        response = requests.get(remote_url, timeout=30)
        response.raise_for_status()

        # 保存文件
        with open(save_path, 'wb') as f:
            f.write(response.content)

        return save_path

    except Exception as e:
        show_auto_hide_message(None, "错误", f"下载失败: {str(e)}", "error")
        return None


def call_api_and_download_result(api_url, headers, request_data, output_dir="output"):
    """调用 API 并下载结果文件

    Args:
        api_url: API 地址
        headers: 请求头
        request_data: 请求数据
        output_dir: 结果保存目录

    Returns:
        tuple: (success, result_data, local_file_path)
    """
    try:
        # 发送 API 请求
        response = requests.post(
            api_url,
            headers=headers,
            json=request_data,
            timeout=180  # 3分钟超时
        )
        response.raise_for_status()
        result = response.json()

        # 检查响应状态
        if result.get("status") == "Success" and result.get("outputs"):
            # 获取结果文件 URL
            result_url = result["outputs"][0].get("object_url", "")

            if result_url:
                # 下载结果文件
                local_path = download_remote_file(result_url, output_dir)

                if local_path:
                    # 解析结果文件
                    with open(local_path, 'r', encoding='utf-8') as f:
                        result_data = json.load(f)

                    return True, result_data, local_path

        return False, None, None

    except Exception as e:
        show_auto_hide_message(None, "错误", f"API 调用失败: {str(e)}", "error")
        return False, None, None
```

### 配置文件管理

基于 JSON 的配置文件管理系统。

#### 实现代码

```python
import json
import os

class ConfigManager:
    """配置文件管理器"""

    def __init__(self, config_file="config.json"):
        self.config_file = config_file
        self.config = self.load_config()

    def load_config(self):
        """加载配置文件"""
        if os.path.exists(self.config_file):
            try:
                with open(self.config_file, 'r', encoding='utf-8') as f:
                    return json.load(f)
            except Exception as e:
                logger.error(f"加载配置失败: {e}")

        # 返回默认配置
        return self.get_default_config()

    def get_default_config(self):
        """获取默认配置"""
        return {
            "api_key": "",
            "model": "default_model",
            "timeout": 180,
            "max_retries": 3
        }

    def save_config(self):
        """保存配置文件"""
        try:
            with open(self.config_file, 'w', encoding='utf-8') as f:
                json.dump(self.config, f, indent=4, ensure_ascii=False)
            return True
        except Exception as e:
            logger.error(f"保存配置失败: {e}")
            return False

    def get(self, key, default=None):
        """获取配置值"""
        return self.config.get(key, default)

    def set(self, key, value):
        """设置配置值"""
        self.config[key] = value
        return self.save_config()

    def update(self, updates):
        """批量更新配置"""
        self.config.update(updates)
        return self.save_config()
```

    "api_key": "new_key",

    "timeout": 300

### 历史记录管理

数据持久化与查询系统。

import json
import os
from datetime import datetime

class HistoryManager:
    """历史记录管理器"""

    def __init__(self, history_file="history.json"):
        self.history_file = history_file
        self.history = self.load_history()

    def load_history(self):
        """加载历史记录"""
        if os.path.exists(self.history_file):
            try:
                with open(self.history_file, 'r', encoding='utf-8') as f:
                    return json.load(f)
            except Exception as e:
                logger.error(f"加载历史记录失败: {e}")
        return []

    def save_history(self):
        """保存历史记录"""
        try:
            # 只保留最近 100 条记录
            if len(self.history) > 100:
                self.history = self.history[-100:]

            with open(self.history_file, 'w', encoding='utf-8') as f:
                json.dump(self.history, f, indent=4, ensure_ascii=False)
            return True
        except Exception as e:
            logger.error(f"保存历史记录失败: {e}")
            return False

    def add_record(self, **kwargs):
        """添加历史记录

        Args:
            **kwargs: 记录字段，如 title, content, url, result 等
        """
        record = {
            "id": str(int(datetime.now().timestamp() * 1000)),
            "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            **kwargs
        }
        self.history.insert(0, record)  # 插入到开头
        self.save_history()

    def delete_record(self, record_id):
        """删除历史记录"""
        self.history = [r for r in self.history if r.get("id") != record_id]
        self.save_history()

    def clear_all(self):
        """清空所有记录"""
        self.history.clear()
        self.save_history()

    def get_history(self):
        """获取所有历史记录"""
        return self.history

    def search(self, keyword):
        """搜索历史记录"""
        keyword = keyword.lower()
        return [r for r in self.history if keyword in str(r).lower()]
```

from PyQt5.QtCore import QThread, pyqtSignal

class WorkerThread(QThread):
    """工作线程基类"""

    # 定义信号
    progress_updated = pyqtSignal(str)      # 进度更新
    finished = pyqtSignal(bool, dict)        # 完成信号
    error_occurred = pyqtSignal(str)         # 错误信号
    log_message = pyqtSignal(str)            # 日志消息

    def __init__(self):
        super().__init__()
        self.is_cancelled = False

    def cancel(self):
        """取消任务"""
        self.is_cancelled = True

    def _log(self, message):
        """内部日志方法"""
        self.log_message.emit(message)

    def run(self):
        """运行任务（子类重写）"""
        try:
            self._log("任务开始...")
            self.progress_updated.emit("正在处理...")

            # 检查是否取消
            if self.is_cancelled:
                self._log("任务已取消")
                return

            # 执行具体任务
            result = self.do_work()

            self._log("任务完成")
            self.finished.emit(True, result)

        except Exception as e:
            self._log(f"任务失败: {str(e)}")
            self.error_occurred.emit(str(e))
            self.finished.emit(False, {})

    def do_work(self):
        """执行具体工作（子类实现）"""
        # 子类重写此方法
        return {}


class MyWorkerThread(WorkerThread):
    """自定义工作线程"""

    def __init__(self, param1, param2):
        super().__init__()
        self.param1 = param1
        self.param2 = param2

    def do_work(self):
        """执行具体工作"""
        # 模拟耗时操作
        import time
        for i in range(10):
            if self.is_cancelled:
                return None

            time.sleep(0.5)
            progress = int((i + 1) / 10 * 100)
            self.progress_updated.emit(f"进度: {progress}%")

        return {"result": "success", "data": [1, 2, 3]}
```

from datetime import datetime
from PyQt5.QtWidgets import QTextEdit

class Logger:
    """日志记录器"""

    def __init__(self, log_widget=None):
        self.log_widget = log_widget
        self.logs = []

    def log(self, message):
        """记录日志"""
        timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        log_entry = f"[{timestamp}] {message}"
        self.logs.append(log_entry)

        # 如果有日志控件，更新显示
        if self.log_widget:
            self.log_widget.append(log_entry)
            # 自动滚动到底部
            scrollbar = self.log_widget.verticalScrollBar()
            scrollbar.setValue(scrollbar.maximum())

    def clear(self):
        """清空日志"""
        self.logs.clear()
        if self.log_widget:
            self.log_widget.clear()

    def get_logs(self):
        """获取所有日志"""
        return self.logs

    def save_to_file(self, file_path):
        """保存日志到文件"""
        try:
            with open(file_path, 'w', encoding='utf-8') as f:
                f.write('\n'.join(self.logs))
            return True
        except Exception as e:
            print(f"保存日志失败: {e}")
            return False
