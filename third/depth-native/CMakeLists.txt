cmake_minimum_required(VERSION 3.20)
project(depth_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORT_ROOT "" CACHE PATH "Extracted ONNX Runtime C/C++ distribution")
# Resolve again when ORT_ROOT changes so headers and the binary share one API.
unset(ORT_LIBRARY CACHE)
find_library(ORT_LIBRARY NAMES onnxruntime PATHS "${ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(depth SHARED depth.cpp depth_io.cpp)
target_include_directories(depth PRIVATE "${ORT_ROOT}/include")
target_compile_definitions(depth PRIVATE DEPTH_BUILD)
target_link_libraries(depth PRIVATE "${ORT_LIBRARY}")
set_target_properties(depth PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_executable(reveric-depth-native main.cpp)
target_link_libraries(reveric-depth-native PRIVATE depth)
if(APPLE)
  set_target_properties(depth PROPERTIES INSTALL_RPATH "@loader_path")
  set_target_properties(reveric-depth-native PROPERTIES INSTALL_RPATH "@loader_path/../lib")
elseif(UNIX)
  set_target_properties(depth PROPERTIES INSTALL_RPATH "$ORIGIN")
  set_target_properties(reveric-depth-native PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(CODE "file(REMOVE \"\${CMAKE_INSTALL_PREFIX}/bin/depth-native${CMAKE_EXECUTABLE_SUFFIX}\")")
install(TARGETS depth reveric-depth-native RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
if(WIN32)
  install(DIRECTORY "${ORT_ROOT}/lib/" DESTINATION bin FILES_MATCHING PATTERN "*.dll")
else()
  install(DIRECTORY "${ORT_ROOT}/lib/" DESTINATION lib FILES_MATCHING PATTERN "*.dylib" PATTERN "*.so*")
endif()
install(FILES "${ORT_ROOT}/LICENSE" "${ORT_ROOT}/ThirdPartyNotices.txt" DESTINATION licenses/onnxruntime)
install(FILES LICENSE.vda DESTINATION licenses)
include(CTest)
if(BUILD_TESTING)
  add_executable(depth-io-test depth_io_test.cpp depth_io.cpp)
  add_test(NAME depth-io COMMAND depth-io-test)
endif()
