## 概述

本教程说明如何将新的独立界面（如 `news.py`）整合到 `storyboard_generator.py` 主程序中，使其成为左侧导航菜单的一个子页面。

### 核心原则

- ✅ **使用 QWidget 类**：新界面应该是继承自 `QWidget` 的页面类，**不是**独立的 `QMainWindow`
- ✅ **接收 parent 参数**：构造函数必须接收 `parent` 参数，用于访问主窗口
- ❌ **不要创建独立 QApplication**：不要在自己的文件中创建 `QApplication` 和事件循环

## 新界面标准结构

### 基础模板（推荐）

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
BOZO-MCN 新闻界面模块
"""
import os
import sys
from PyQt5.QtWidgets import QWidget, QVBoxLayout, QLabel
from PyQt5.QtCore import Qt
from qfluentwidgets import (FluentIcon, SubtitleLabel, BodyLabel,
                          PushButton, CardWidget, InfoBar, InfoBarPosition)


class NewsPage(QWidget):
    """新闻页面 - 可集成到主程序的子页面"""

    def __init__(self, parent=None):
        super().__init__(parent)
        self.main_window = parent  # 保存主窗口引用，如需访问
        self.init_ui()

    def init_ui(self):
        """初始化界面"""
        layout = QVBoxLayout(self)
        layout.setSpacing(15)
        layout.setContentsMargins(20, 20, 20, 20)

        # 标题
        title = SubtitleLabel("📰 新闻资讯")
        title.setStyleSheet("font-size: 24px; font-weight: bold;")
        layout.addWidget(title)

        # 内容卡片
        content_card = CardWidget()
        content_layout = QVBoxLayout(content_card)

        content_label = BodyLabel("这里是新闻内容区域...")
        content_layout.addWidget(content_label)

        layout.addWidget(content_card)
        layout.addStretch()


# ==================== 以下代码仅用于独立测试 ====================
# 当作为主程序运行时使用，集成时会被忽略
class TestMainWindow(QWidget):
    """测试用主窗口 - 仅用于独立运行测试"""
    def __init__(self):
        super().__init__()
        self.setWindowTitle("新闻界面测试")
        self.setMinimumSize(800, 600)
        layout = QVBoxLayout(self)
        layout.addWidget(NewsPage(self))


def main():
    """仅用于独立测试的入口函数"""
    from PyQt5.QtWidgets import QApplication
    from qfluentwidgets import setTheme, Theme

    app = QApplication(sys.argv)
    setTheme(Theme.DARK)
    window = TestMainWindow()
    window.show()
    sys.exit(app.exec_())


if __name__ == "__main__":
    main()
```

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
BOZO-MCN 新闻资讯页面
"""
import os
import sys
import json
import requests
from datetime import datetime
from PyQt5.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout, QLabel,
                            QTextEdit, QScrollArea, QFrame)
from PyQt5.QtCore import Qt, QThread, pyqtSignal, QUrl
from PyQt5.QtGui import QDesktopServices
from qfluentwidgets import (FluentIcon, SubtitleLabel, BodyLabel,
                          PushButton, CardWidget, InfoBar, InfoBarPosition,
                          ProgressBar, SmoothScrollArea)


class NewsPage(SmoothScrollArea):
    """新闻资讯页面"""

    def __init__(self, parent=None):
        super().__init__(parent)
        self.main_window = parent
        self.init_ui()

    def init_ui(self):
        """初始化界面"""
        # 创建主容器
        container = QWidget()
        layout = QVBoxLayout(container)
        layout.setSpacing(20)
        layout.setContentsMargins(30, 30, 30, 30)

        # 标题栏
        title_layout = QHBoxLayout()
        title = SubtitleLabel("📰 新闻资讯")
        title.setStyleSheet("font-size: 28px; font-weight: bold;")
        title_layout.addWidget(title)
        title_layout.addStretch()

        # 刷新按钮
        self.refresh_btn = PushButton(FluentIcon.SYNC, "刷新")
        self.refresh_btn.clicked.connect(self.refresh_news)
        title_layout.addWidget(self.refresh_btn)

        layout.addLayout(title_layout)

        # 新闻列表容器
        self.news_container = QWidget()
        self.news_layout = QVBoxLayout(self.news_container)
        self.news_layout.setSpacing(15)

        # 加载提示
        self.loading_label = BodyLabel("正在加载新闻...")
        self.news_layout.addWidget(self.loading_label)

        layout.addWidget(self.news_container)
        layout.addStretch()

        # 设置为滚动区域的组件
        self.setWidget(container)
        self.setWidgetResizable(True)

        # 自动加载新闻
        self.refresh_news()

    def refresh_news(self):
        """刷新新闻"""
        self.loading_label.setText("正在加载新闻...")
        # TODO: 实现新闻加载逻辑
        # 这里可以添加实际的 API 调用或数据加载

        # 示例：添加模拟新闻
        self.clear_news()
        self.add_news_item(
            "BOZO-MCN v2.0 发布",
            "全新分镜脚本生成功能上线",
            "2026-01-12"
        )
        self.add_news_item(
            "AI 模型更新",
            "新增多个高质量绘图模型",
            "2026-01-10"
        )

    def clear_news(self):
        """清空新闻列表"""
        while self.news_layout.count():
            item = self.news_layout.takeAt(0)
            if item.widget():
                item.widget().deleteLater()

    def add_news_item(self, title, content, date):
        """添加新闻条目"""
        card = CardWidget()
        card_layout = QVBoxLayout(card)

        # 标题
        title_label = SubtitleLabel(title)
        title_label.setStyleSheet("font-size: 18px; font-weight: bold;")
        card_layout.addWidget(title_label)

        # 内容
        content_label = BodyLabel(content)
        content_label.setWordWrap(True)
        card_layout.addWidget(content_label)

        # 日期
        date_label = BodyLabel(date)
        date_label.setStyleSheet("color: #999; font-size: 12px;")
        card_layout.addWidget(date_label)

        self.news_layout.addWidget(card)


# ==================== 独立测试代码 ====================

## 常见问题

### Q1: 导入时报 `ImportError`

**原因**：文件不在同一目录或模块名称错误

**解决**：
```python
# 确认文件路径
import os
print(os.path.listdir('.'))  # 查看当前目录文件

# 检查文件名是否正确
from news import NewsPage  # 确保 news.py 存在
```

### Q2: 页面显示空白

**原因**：界面初始化不完整或布局未正确设置

**解决**：
```python
# 确保调用了父类初始化
super().__init__(parent)

# 确保设置了布局
self.setLayout(layout)

# 如果继承自 ScrollArea，确保设置 widget
self.setWidget(container)
self.setWidgetResizable(True)
```

### Q3: 点击导航没反应

**原因**：`setObjectName()` 未设置或设置不正确

**解决**：
```python
self.news_page = NewsPage(self)
self.news_page.setObjectName("news_page")  # 必须设置唯一的对象名
```

### Q4: 如何访问主窗口的方法？

**通过 parent 参数访问**：
```python
class NewsPage(QWidget):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.main_window = parent  # 保存主窗口引用

    def some_function(self):
        # 访问主窗口的方法
        if hasattr(self.main_window, 'open_directory'):
            self.main_window.open_directory('/path/to/dir')
```

### Q5: 如何在页面间传递数据？

**方法 1：通过主窗口中转**
```python
# 在主窗口中定义数据存储
class MainWindow(FluentWindow):
    def __init__(self):
        super().__init__()
        self.shared_data = {}

    def set_data(self, key, value):
        self.shared_data[key] = value

# 在子页面中访问
self.main_window.shared_data['key'] = value
```

**方法 2：使用信号槽**
```python
class NewsPage(QWidget):
    # 定义信号
    data_updated = pyqtSignal(dict)

    def send_data(self):
        self.data_updated.emit({'key': 'value'})

# 在主窗口中连接
self.news_page.data_updated.connect(self.handle_data_update)
```
